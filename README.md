


# Employee & HR Analytics Data Engineering Project

## Project Overview

This project implements an Employee and HR Analytics Data Warehouse using Python, Snowflake SQL, and Streamlit.

The objective is to collect HR data from multiple CSV files, perform data quality checks, transform employee information, maintain historical records, and generate HR business insights.

## Technologies Used

- Python
- Snowflake
- Snowflake SQL
- Snowflake Streams and Tasks
- SCD Type 2
- Streamlit
- Pandas

   CREATE OR REPLACE TABLE HR_ANALYTICS_DB.STAGING.STG_EMPLOYEES AS
SELECT
    EMPLOYEE_ID,
    TRIM(FIRST_NAME) AS FIRST_NAME,
    TRIM(LAST_NAME) AS LAST_NAME,
    INITCAP(TRIM(DEPARTMENT)) AS DEPARTMENT,
    SALARY,
    HIRE_DATE,
    UPPER(TRIM(EMPLOYMENT_STATUS)) AS EMPLOYMENT_STATUS,
    UPDATED_AT
FROM HR_ANALYTICS_DB.RAW.RAW_EMPLOYEES
WHERE EMPLOYEE_ID IS NOT NULL
QUALIFY ROW_NUMBER() OVER (
    PARTITION BY EMPLOYEE_ID
    ORDER BY UPDATED_AT DESC NULLS LAST, INGESTED_AT DESC
) = 1;




CREATE TABLE IF NOT EXISTS
HR_ANALYTICS_DB.CURATED.DIM_EMPLOYEE (
    EMPLOYEE_ID NUMBER,
    FIRST_NAME VARCHAR,
    LAST_NAME VARCHAR,
    DEPARTMENT VARCHAR,
    SALARY NUMBER(12,2),
    HIRE_DATE DATE,
    EMPLOYMENT_STATUS VARCHAR,

    EFFECTIVE_TO TIMESTAMP_NTZ,

);







INSERT INTO HR_ANALYTICS_DB.CURATED.DIM_EMPLOYEE
SELECT
    EMPLOYEE_ID,
    FIRST_NAME,
    LAST_NAME,
    DEPARTMENT,
    SALARY,
    HIRE_DATE,
    EMPLOYMENT_STATUS,
    CURRENT_TIMESTAMP(),
    '9999-12-31'::TIMESTAMP_NTZ,
    TRUE
FROM HR_ANALYTICS_DB.STAGING.STG_EMPLOYEES;



CREATE OR REPLACE STREAM HR_ANALYTICS_DB.RAW.EMPLOYEE_STREAM
ON TABLE HR_ANALYTICS_DB.RAW.RAW_EMPLOYEES
APPEND_ONLY = TRUE;






CREATE TABLE IF NOT EXISTS
    HR_ANALYTICS_DB.CURATED.EMPLOYEE_CURRENT (
        EMPLOYEE_ID NUMBER,
        DEPARTMENT VARCHAR,
        SALARY NUMBER(12,2),
        EMPLOYMENT_STATUS VARCHAR
    );










MERGE INTO HR_ANALYTICS_DB.CURATED.EMPLOYEE_CURRENT T
USING HR_ANALYTICS_DB.STAGING.STG_EMPLOYEES S
   ON T.EMPLOYEE_ID = S.EMPLOYEE_ID
WHEN MATCHED THEN UPDATE SET
    T.DEPARTMENT = S.DEPARTMENT,
    T.SALARY = S.SALARY,
    T.EMPLOYMENT_STATUS = S.EMPLOYMENT_STATUS
WHEN NOT MATCHED THEN INSERT (
    EMPLOYEE_ID, DEPARTMENT, SALARY, EMPLOYMENT_STATUS
)
VALUES (
    S.EMPLOYEE_ID, S.DEPARTMENT, S.SALARY, S.EMPLOYMENT_STATUS
);

CREATE OR REPLACE VIEW
HR_ANALYTICS_DB.REPORTING.VW_DEPARTMENT_HEADCOUNT AS
SELECT
    DEPARTMENT,
    COUNT(*) AS EMPLOYEE_COUNT
FROM HR_ANALYTICS_DB.CURATED.DIM_EMPLOYEE
WHERE IS_CURRENT = TRUE
  AND EMPLOYMENT_STATUS = 'ACTIVE'
GROUP BY DEPARTMENT;






CREATE OR REPLACE VIEW
HR_ANALYTICS_DB.REPORTING.VW_DEPARTMENT_SALARY AS
SELECT
    DEPARTMENT,
    ROUND(AVG(SALARY), 2) AS AVG_SALARY,
    ROUND(MIN(SALARY), 2) AS MIN_SALARY,
    ROUND(MAX(SALARY), 2) AS MAX_SALARY
FROM HR_ANALYTICS_DB.CURATED.DIM_EMPLOYEE
WHERE IS_CURRENT = TRUE
  AND EMPLOYMENT_STATUS = 'ACTIVE'
GROUP BY DEPARTMENT;




CREATE OR REPLACE VIEW
HR_ANALYTICS_DB.REPORTING.VW_MONTHLY_HIRING AS
SELECT
    DATE_TRUNC('MONTH', HIRE_DATE)::DATE AS HIRE_MONTH,
    COUNT(DISTINCT EMPLOYEE_ID) AS HIRES
FROM HR_ANALYTICS_DB.CURATED.DIM_EMPLOYEE
GROUP BY 1;
